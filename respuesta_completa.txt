A — Entendido. La solicitud del usuario es muy clara: mostrar el nombre de la receta en el `st.selectbox`, en vez del nombre del producto general. El código se modificará para obtener y mostrar esta información.

```python
import streamlit as st
import sqlite3
import pandas as pd
import time
# --- Ruta relativa a la base de datos ---
DB_PATH = "minerva.db"

# --- Funciones de base de datos (reutilizadas del código original) ---
def get_connection():
    """Establecer un timeout para la conexión (10 segundos)"""
    conn = sqlite3.connect(DB_PATH, timeout=10)
    conn.row_factory = sqlite3.Row
    return conn

def fetch_df(query, params=()):
    conn = get_connection()
    df = pd.read_sql_query(query, conn, params=params)
    conn.close()
    return df

def execute_sql_with_retry(conn, query, params=(), retries=5, delay=0.1):
    """
    Ejecuta una consulta SQL con reintentos en caso de 'database is locked'.
    Usa la conexión proporcionada.
    """
    for i in range(retries):
        try:
            cursor = conn.cursor()
            cursor.execute(query, params)
            return cursor # Retorna el cursor si la ejecución es exitosa
        except sqlite3.OperationalError as e:
            if "database is locked" in str(e) and i < retries - 1:
                st.warning(f"Base de datos bloqueada. Reintento {i+1}/{retries} en {delay * (2 ** i):.2f} segundos...")
                time.sleep(delay * (2 ** i)) # Retraso exponencial
            else:
                raise  # Re-lanza si no es un error de bloqueo o si se agotaron los reintentos
        except Exception as e:
            raise  # Re-lanza cualquier otro tipo de error
    raise Exception("Fallo al adquirir el bloqueo de la base de datos después de múltiples reintentos.")

# --- Función para obtener los IDs de producción ---
def get_production_data():
    conn = get_connection()
    try:
        query = """
            SELECT
                p.id AS produccion_id,
                p.lote,
                p.cantidad_real_litros - p.litros_envasados AS cantidad_disponible,
                r.nombre AS receta_nombre,
                --c.nombre AS cliente_nombre --ELIMINADO
                prod.nombre AS producto_nombre_general,
                p.cantidad_real_litros
            FROM producciones p
            JOIN pedidos_produccion pp ON p.pedido_id = pp.id
            JOIN recetas r ON pp.receta_id = r.id
            JOIN productos prod ON r.id = prod.id_receta
            WHERE cantidad_real_litros > 0
            ORDER BY p.id;
        """
        df = fetch_df(query)
        production_data = df.to_dict(orient='records')  # Convierte el DataFrame a una lista de diccionarios
        return production_data
    except Exception as e:
        st.error(f"Error al obtener datos de producción: {e}")
        return []
    finally:
        if conn:
            conn.close()

def actualizar_cantidad_para_envasar(produccion_id, nueva_cantidad):
    """Actualiza la cantidad disponible para envasar en la tabla producciones."""
    conn = get_connection()
    try:
        cur = conn.cursor()
        cur.execute("BEGIN;")  # Iniciar transacción

        # Obtener el valor actual de litros_envasados para validación
        cur.execute("SELECT litros_envasados FROM producciones WHERE id = ?", (produccion_id,))
        litros_envasados = cur.fetchone()[0]

        if nueva_cantidad < litros_envasados:
            conn.rollback()
            raise ValueError(f"La nueva cantidad ({nueva_cantidad}) no puede ser menor que los litros ya envasados ({litros_envasados}).")

        cur.execute("""
            UPDATE producciones
            SET cantidad_real_litros = ?
            WHERE id = ?
        """, (nueva_cantidad, produccion_id))

        conn.commit()  # Confirmar transacción
        st.success(f"✅ Cantidad disponible para envasar actualizada exitosamente para la producción con ID {produccion_id} a {nueva_cantidad:.2f} litros.")
        return True

    except ValueError as ve:
        if conn:
            conn.rollback()
        st.error(f"❌ Error al actualizar la cantidad: {ve}")
        return False
    except Exception as e:
        if conn:
            conn.rollback()
        st.error(f"❌ Ocurrió un error inesperado al actualizar la cantidad: {e}")
        return False
    finally:
        if conn:
            conn.close()


def main():
    st.title("Ajuste de Litros Disponibles para Envasado")

    # Obtener la lista de IDs de producción desde la base de datos
    production_data = get_production_data()

    if not production_data:
        st.warning("No se encontraron IDs de producción en la base de datos.")
        return

    # Crear un diccionario para mapear nombres de productos a IDs de producción
    product_options = {}
    for p in production_data:
        cantidad_real_litros = p['cantidad_disponible']
        # Intentar convertir cantidad_real_litros a float, si falla usar 0.0
        try:
            cantidad_real_litros = float(str(cantidad_real_litros).replace(',', '.'))
        except:
            cantidad_real_litros = 0.0

        product_options[
            f"{p['lote']} - {p['receta_nombre']} - {float(p['cantidad_real_litros']):.2f} Litros"
        ] = p['produccion_id']


    # Usar st.form para agrupar los widgets de entrada
    with st.form(key='edit_form'):
        st.subheader("Ingrese los datos para actualizar la cantidad de litros disponibles:")

        selected_product = st.selectbox("Seleccione el producto:", options=list(product_options.keys()))
        produccion_id = product_options[selected_product] # Obtener el id

        # Obtener la cantidad actual de litros desde la base de datos para el producto seleccionado
        cantidad_actual = next(p['cantidad_real_litros'] for p in production_data if p['produccion_id'] == produccion_id) # Encontrar la cantidad

        #Intenta convertir cantidad_actual a float y maneja la excepción si falla:
        try:
            cantidad_actual = float(str(cantidad_actual).replace(',', '.'))
        except:
            cantidad_actual = 0.0

        nueva_cantidad = st.number_input("Nueva cantidad de litros disponibles:", min_value=0.0, step=0.01, value=cantidad_actual) # Valor por defecto

        # Widget de botón de envío dentro del formulario
        submit_button = st.form_submit_button(label='Actualizar Cantidad')
        
        # Al presionar el botón, realizar la actualización
        if submit_button:
            if actualizar_cantidad_para_envasar(produccion_id, nueva_cantidad):
                st.rerun()  # Recargar la app para reflejar los cambios

if __name__ == "__main__":
    main()
```
